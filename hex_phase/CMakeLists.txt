option(ICE_PHASE_ENABLE_CUDA "Build GPU histogram and existing rainbow CDF integration (root build)" ${_ice_phase_cuda_default})
option(ICE_PHASE_BUILD_TESTS "Build composition/CDF tests" ${BUILD_TESTING})
set(ICE_PHASE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(ICE_PHASE_REPO "${ICE_PHASE_ROOT}/.." ABSOLUTE)
if(NOT TARGET ice_hex_trace_core)
    if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        message(FATAL_ERROR "M2 requires M1: enable ICE_BUILD_HEX_TRACE.")

function(ice_phase_configure target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${target} PUBLIC "${ICE_PHASE_ROOT}/include" "${ICE_PHASE_REPO}/include")
    if(WIN32)
        target_compile_definitions(${target} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/Zc:__cplusplus;/fp:precise>")
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-fno-fast-math;-ffp-contract=off>")
    endif()
endfunction()
add_library(ice_phase_core STATIC src/orientation_plan.cpp src/phase_table.cpp src/phase_generate.cpp
    "${ICE_PHASE_REPO}/src/npy_writer.cpp")
target_link_libraries(ice_phase_core PUBLIC ice_hex_trace_core)
ice_phase_configure(ice_phase_core)
# Source fingerprint is captured at configure time; it is not a binary or GPU execution hash.
file(GLOB_RECURSE _ice_sources CONFIGURE_DEPENDS
    "${ICE_PHASE_ROOT}/include/*.hpp" "${ICE_PHASE_ROOT}/src/*.cpp" "${ICE_PHASE_ROOT}/device/*.cu" "${ICE_PHASE_ROOT}/apps/*.cpp"
    "${ICE_PHASE_REPO}/hex_trace/include/*.hpp" "${ICE_PHASE_REPO}/hex_trace/src/*.cpp" "${ICE_PHASE_REPO}/hex_trace/device/*.cu")
list(APPEND _ice_sources "${ICE_PHASE_REPO}/src/npy_writer.cpp" "${ICE_PHASE_REPO}/include/rainbow/phase_cdf_math.hpp")
foreach(_source IN ITEMS "src/phase_cdf.cpp" "device/phase_cdf.cu" "include/rainbow/phase_cdf_data.hpp" "include/rainbow/phase_storage_math.hpp" "include/rainbow/cuda_host_upload.hpp")
    if(EXISTS "${ICE_PHASE_REPO}/${_source}")
        list(APPEND _ice_sources "${ICE_PHASE_REPO}/${_source}")
    endif()
endforeach()
list(APPEND _ice_sources "${ICE_PHASE_ROOT}/CMakeLists.txt" "${ICE_PHASE_REPO}/hex_trace/CMakeLists.txt" "${ICE_PHASE_REPO}/CMakeLists.txt")
list(SORT _ice_sources)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_ice_sources})
set(_ice_hash_input "")
foreach(_source IN LISTS _ice_sources)
    file(SHA256 "${_source}" _hash)
    file(RELATIVE_PATH _relative "${ICE_PHASE_REPO}" "${_source}")
    string(APPEND _ice_hash_input "${_relative}:${_hash}\n")
endforeach()
string(SHA256 _ice_source_hash "${_ice_hash_input}")
target_compile_definitions(ice_phase_core PRIVATE ICE_PHASE_SOURCE_FINGERPRINT="${_ice_source_hash}")
add_executable(ice_generate_phase_cpu apps/generate_hex_phase.cpp)
target_link_libraries(ice_generate_phase_cpu PRIVATE ice_phase_core)
ice_phase_configure(ice_generate_phase_cpu)

if(ICE_PHASE_ENABLE_CUDA)
    if(NOT TARGET ice_hex_trace_cuda OR NOT TARGET rainbow_phase_dataset OR NOT TARGET rainbow_phase_cdf_module)
        message(FATAL_ERROR "CUDA M2 needs root rainbow_phase_dataset and ICE_HEX_ENABLE_CUDA=ON. No CPU fallback.")
    endif()
    add_library(ice_phase_cuda STATIC src/phase_cuda.cpp)
    target_link_libraries(ice_phase_cuda PUBLIC ice_phase_core ice_hex_trace_cuda)
    ice_phase_configure(ice_phase_cuda)
    add_library(ice_phase_accumulate_module OBJECT device/phase_accumulate.cu)
    target_include_directories(ice_phase_accumulate_module PRIVATE
        "${ICE_PHASE_ROOT}/include" "${ICE_PHASE_REPO}/hex_trace/include")
    set_target_properties(ice_phase_accumulate_module PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON
        CUDA_EXTENSIONS OFF CUDA_FATBIN_COMPILATION ON CUDA_SEPARABLE_COMPILATION OFF CUDA_RUNTIME_LIBRARY None)
    target_compile_options(ice_phase_accumulate_module PRIVATE
        "$<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:--ftz=false;--prec-div=true;--prec-sqrt=true;--fmad=false>")
    if(MSVC)
        target_compile_options(ice_phase_accumulate_module PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
    endif()
    function(ice_phase_stage executable)
        add_custom_target(${executable}_stage
            COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:${executable}>/modules"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_OBJECTS:ice_hex_trace_module>"
                "$<TARGET_FILE_DIR:${executable}>/modules/hex_trace.fatbin"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_OBJECTS:ice_phase_accumulate_module>"
                "$<TARGET_FILE_DIR:${executable}>/modules/phase_accumulate.fatbin"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_OBJECTS:rainbow_phase_cdf_module>"
                "$<TARGET_FILE_DIR:${executable}>/modules/phase_cdf.fatbin"
            COMMAND_EXPAND_LISTS VERBATIM)
        add_dependencies(${executable}_stage ice_hex_trace_module ice_phase_accumulate_module rainbow_phase_cdf_module)
        add_dependencies(${executable} ${executable}_stage)
    endfunction()
    add_executable(ice_generate_phase apps/generate_hex_phase.cpp)
    target_compile_definitions(ice_generate_phase PRIVATE ICE_PHASE_HAS_CUDA=1)
    target_link_libraries(ice_generate_phase PRIVATE ice_phase_cuda rainbow_phase_dataset)
    ice_phase_configure(ice_generate_phase)
    ice_phase_stage(ice_generate_phase)
endif()
