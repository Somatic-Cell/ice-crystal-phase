cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(IceDeterministicAtlas LANGUAGES CXX)
    include(CTest)
    set(_atlas_cuda_default OFF)
else()
    set(_atlas_cuda_default ${ICE_HEX_ENABLE_CUDA})
endif()
option(ICE_ATLAS_ENABLE_CUDA "Build deterministic CUDA area-ray precomputation" ${_atlas_cuda_default})
option(ICE_ATLAS_BUILD_TESTS "Build atlas and deterministic composition tests" ${BUILD_TESTING})
set(ICE_ATLAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(ICE_ATLAS_REPO "${ICE_ATLAS_ROOT}/.." ABSOLUTE)
if(NOT TARGET ice_hex_trace_core)
    if(NOT DEFINED ICE_HEX_ENABLE_CUDA)
        set(ICE_HEX_ENABLE_CUDA ${ICE_ATLAS_ENABLE_CUDA} CACHE BOOL "M1 dependency CUDA")
    endif()
    add_subdirectory("${ICE_ATLAS_REPO}/hex_trace" "hex_trace_dependency")
endif()
function(ice_atlas_configure target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${target} PUBLIC "${ICE_ATLAS_ROOT}/include"
        "${ICE_ATLAS_REPO}/hex_trace/include" "${ICE_ATLAS_REPO}/hex_phase/include" "${ICE_ATLAS_REPO}/include")
    if(WIN32)
        target_compile_definitions(${target} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/Zc:__cplusplus;/fp:precise>")
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-fno-fast-math;-ffp-contract=off>")
    endif()
endfunction()
add_library(ice_atlas_core STATIC src/atlas.cpp "${ICE_ATLAS_REPO}/src/npy_writer.cpp")
ice_atlas_configure(ice_atlas_core)
target_link_libraries(ice_atlas_core PUBLIC ice_hex_trace_core)
file(GLOB_RECURSE _atlas_sources CONFIGURE_DEPENDS "${ICE_ATLAS_ROOT}/include/*.hpp"
    "${ICE_ATLAS_ROOT}/src/*.cpp" "${ICE_ATLAS_ROOT}/device/*.cu" "${ICE_ATLAS_ROOT}/apps/*.cpp"
    "${ICE_ATLAS_REPO}/hex_trace/include/*.hpp")
list(APPEND _atlas_sources "${ICE_ATLAS_ROOT}/CMakeLists.txt" "${ICE_ATLAS_REPO}/hex_trace/src/hex_trace.cpp"
    "${ICE_ATLAS_REPO}/hex_phase/include/ice_crystal/phase_math.hpp" "${ICE_ATLAS_REPO}/src/npy_writer.cpp")
list(SORT _atlas_sources)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_atlas_sources})
set(_atlas_hash_input "")
foreach(_s IN LISTS _atlas_sources)
    file(SHA256 "${_s}" _h)
    file(RELATIVE_PATH _r "${ICE_ATLAS_REPO}" "${_s}")
    string(APPEND _atlas_hash_input "${_r}:${_h}\n")
endforeach()
string(SHA256 _atlas_hash "${_atlas_hash_input}")
add_executable(ice_precompute_atlas_cpu apps/precompute_atlas.cpp)
ice_atlas_configure(ice_precompute_atlas_cpu)
target_link_libraries(ice_precompute_atlas_cpu PRIVATE ice_atlas_core)
target_compile_definitions(ice_precompute_atlas_cpu PRIVATE ICE_ATLAS_SOURCE_FINGERPRINT="${_atlas_hash}")
if(ICE_ATLAS_ENABLE_CUDA)
    if(NOT TARGET ice_hex_trace_cuda)
        message(FATAL_ERROR "Atlas CUDA requires ICE_HEX_ENABLE_CUDA=ON. No CPU fallback.")
    endif()
    add_library(ice_atlas_cuda STATIC src/atlas_cuda.cpp)
    ice_atlas_configure(ice_atlas_cuda)
    target_link_libraries(ice_atlas_cuda PUBLIC ice_atlas_core ice_hex_trace_cuda)
    add_library(ice_atlas_module OBJECT device/atlas_trace.cu)
    target_include_directories(ice_atlas_module PRIVATE "${ICE_ATLAS_ROOT}/include"
        "${ICE_ATLAS_REPO}/hex_trace/include" "${ICE_ATLAS_REPO}/hex_phase/include")
    set_target_properties(ice_atlas_module PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_EXTENSIONS OFF
        CUDA_FATBIN_COMPILATION ON CUDA_SEPARABLE_COMPILATION OFF CUDA_RUNTIME_LIBRARY None)
    target_compile_options(ice_atlas_module PRIVATE
        "$<$<COMPILE_LANG_AND_ID:CUDA,NVIDIA>:--ftz=false;--prec-div=true;--prec-sqrt=true;--fmad=false>")
    if(MSVC)
        target_compile_options(ice_atlas_module PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
    endif()
    function(ice_atlas_stage executable)
        add_custom_target(${executable}_stage
            COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:${executable}>/modules"
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_OBJECTS:ice_atlas_module>"
                "$<TARGET_FILE_DIR:${executable}>/modules/atlas_trace.fatbin"
            COMMAND_EXPAND_LISTS VERBATIM)
        add_dependencies(${executable}_stage ice_atlas_module)
        add_dependencies(${executable} ${executable}_stage)
    endfunction()
    add_executable(ice_precompute_atlas apps/precompute_atlas.cpp)
    ice_atlas_configure(ice_precompute_atlas)
    target_compile_definitions(ice_precompute_atlas PRIVATE ICE_ATLAS_HAS_CUDA=1 ICE_ATLAS_SOURCE_FINGERPRINT="${_atlas_hash}")
    target_link_libraries(ice_precompute_atlas PRIVATE ice_atlas_cuda)
    ice_atlas_stage(ice_precompute_atlas)
endif()

            add_test(NAME ice_atlas_polar_azimuth COMMAND "${Python3_EXECUTABLE}" "${ICE_ATLAS_ROOT}/tests/test_polar_azimuth.py")
            add_test(NAME ice_atlas_cli COMMAND "${Python3_EXECUTABLE}" "${ICE_ATLAS_ROOT}/tests/cli_test.py"
                "$<TARGET_FILE:ice_precompute_atlas_cpu>" "${ICE_ATLAS_REPO}")
        endif()
    endif()
    if(ICE_ATLAS_ENABLE_CUDA)
        add_executable(ice_atlas_cuda_tests tests/atlas_cuda_tests.cpp)
        ice_atlas_configure(ice_atlas_cuda_tests)
        target_link_libraries(ice_atlas_cuda_tests PRIVATE ice_atlas_cuda)
        ice_atlas_stage(ice_atlas_cuda_tests)
        add_test(NAME ice_atlas_cuda COMMAND ice_atlas_cuda_tests)
        if(Python3_Interpreter_FOUND AND _atlas_numpy EQUAL 0)
            add_test(NAME ice_atlas_cuda_cli COMMAND "${Python3_EXECUTABLE}" "${ICE_ATLAS_ROOT}/tests/cli_test.py"
                "$<TARGET_FILE:ice_precompute_atlas>" "${ICE_ATLAS_REPO}")
        endif()
    endif()
endif()
